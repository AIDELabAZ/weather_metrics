\begin{landscape}
	\begin{figure}[!htbp]
		\begin{minipage}{\linewidth}
			\caption{Spatial Dependence and First-Stage Instrument Strength}
			\label{app:space_panel}
			{\centering
				\includegraphics[width=.97\linewidth,keepaspectratio]{Figures/space_panel.pdf}\par}
			\smallskip
			\footnotesize \textit{Note}: Panel A shows mean standardized cross-products
			of first-stage scores, not Pearson correlations. Location-year score sums
			are divided by the square root of household count and standardized within
			specification and year; curves take the median of 12 product-outcome
			specification means per metric. Panel B uses year and household FE, with
			the same coefficient and observations for each covariance comparison:
			168 matched specifications per country. The spatial estimator combines
			unrestricted own-household serial covariance with contemporaneous
			cross-household covariance under a 100-km Bartlett kernel. Percentages
			are descriptive threshold-crossing shares, not tests of their differences.
			Geographic coverage restricts the sample, especially in Tanzania and
			Uganda; missing or confidential coordinates are not imputed.
		\end{minipage}
	\end{figure}
\end{landscape}
